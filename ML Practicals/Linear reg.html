mean_x = np.mean(x) 
mean_y = np.mean(y) 
n = len(x) 
numer = 0 
denom = 0 
for i in range(n): 
    numer += (x[i] - mean_x) * (y[i] - mean_y) 
    denom += (x[i] - mean_x) **2
    b1 = numer/denom 
    b0 = mean_y - b1 * mean_x
